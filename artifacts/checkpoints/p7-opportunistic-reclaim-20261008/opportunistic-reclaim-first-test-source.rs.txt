//! P7-016 wiring #9: bounded production drains on real SQLite databases.
//! Synthetic, explicitly aged lease fixtures; these are not crash evidence.
//! The separate unchanged L3 driver supplies real process/clock recovery evidence.
use cc_db::{
    index_db::IndexDb,
    semantic_outbox::{claim_next_on, now_unix},
    semantic_publish::LifecycleFence,
};
use cc_semantic::{
    queue::{
        drain_pending_parallel_with_lifecycle, drain_pending_with_lifecycle, TaskExit, WorkerLimits,
    },
    space_switch::drain_space_revocations,
};
use std::{
    path::PathBuf,
    sync::atomic::{AtomicUsize, Ordering},
    time::Duration,
};

static NEXT: AtomicUsize = AtomicUsize::new(0);
const TARGET: &str = "target";
const FOREIGN: &str = "older-foreign";
const EXPIRED: usize = 137;

struct Fixture {
    root: PathBuf,
    db: IndexDb,
}
impl Fixture {
    fn new(op: &str) -> Self {
        let root = std::env::temp_dir().join(format!(
            "cc-opportunistic-reclaim-{}-{}",
            std::process::id(),
            NEXT.fetch_add(1, Ordering::Relaxed)
        ));
        std::fs::create_dir(&root).unwrap();
        let db = IndexDb::open(&root.join("index.db")).unwrap().0;
        let f = Self { root, db };
        let conn = f.conn();
        conn.execute(
            "INSERT INTO semantic_spaces(space_id,spec_json,state) VALUES(?1,'{}','active')",
            [TARGET],
        )
        .unwrap();
        conn.execute(
            "INSERT INTO semantic_spaces(space_id,spec_json,state) VALUES(?1,'{}','revoked')",
            [FOREIGN],
        )
        .unwrap();
        // Older foreign task IDs exceed three whole 64-row pages. They must
        // neither absorb the target page nor be mutated by its consumer.
        for i in 0..193 {
            f.insert(FOREIGN, &format!("foreign-{i}"), "embed", "claimed", 1000.0);
        }
        for i in 0..EXPIRED {
            f.insert(TARGET, &format!("expired-{i}"), op, "claimed", 1000.0);
        }
        for i in 0..3 {
            f.insert(
                TARGET,
                &format!("unexpired-{i}"),
                op,
                "claimed",
                now_unix() + 3600.0,
            );
        }
        f.insert(TARGET, "terminal-failed", op, "failed", 1000.0);
        f.insert(TARGET, "terminal-superseded", op, "superseded", 1000.0);
        for i in 0..2 {
            f.insert(
                TARGET,
                &format!("future-{i}"),
                op,
                "pending",
                now_unix() + 3600.0,
            );
        }
        // Observe actual row transitions, independently of the production
        // report. A later worker retry has a new owner and is not a reclaim.
        conn.execute_batch(
            "CREATE TABLE reclaim_audit(task_id INTEGER, space_id TEXT,
                 attempts_before INTEGER, attempts_after INTEGER,
                 token_after TEXT, deadline_after REAL, owner_after TEXT);
             CREATE TRIGGER record_reclaim AFTER UPDATE ON semantic_outbox
             WHEN OLD.state='claimed' AND NEW.state='pending'
                  AND OLD.claim_owner='stranded'
             BEGIN
               INSERT INTO reclaim_audit VALUES(OLD.task_id,OLD.space_id,
                 OLD.attempt_count,NEW.attempt_count,NEW.lease_token,
                 NEW.lease_expires_at,NEW.claim_owner);
             END;",
        )
        .unwrap();
        f
    }
    fn conn(&self) -> rusqlite::Connection {
        let c = rusqlite::Connection::open(self.db.admin().db_path()).unwrap();
        c.busy_timeout(Duration::from_secs(5)).unwrap();
        c
    }
    fn insert(&self, space: &str, key: &str, op: &str, state: &str, deadline: f64) {
        self.conn()
            .execute(
                "INSERT INTO semantic_outbox(doc_key,doc_version,input_digest,space_id,op,state,
             attempt_count,lease_token,lease_expires_at,claim_owner,available_at,last_error,
             created_at,updated_at) VALUES(?1,'v1','input',?2,?3,?4,3,?5,?6,'stranded',?7,
             'original failure','2026-01-01','2026-01-01')",
                rusqlite::params![
                    key,
                    space,
                    op,
                    state,
                    format!("original-{key}"),
                    deadline,
                    if state == "pending" { deadline } else { 900.0 }
                ],
            )
            .unwrap();
    }
    fn rows(&self, where_sql: &str) -> Vec<String> {
        let c = self.conn();
        let mut s = c
            .prepare(&format!(
                "SELECT task_id,doc_key,doc_version,input_digest,space_id,op,state,attempt_count,
             lease_token,lease_expires_at,claim_owner,available_at,last_error,created_at,updated_at
             FROM semantic_outbox WHERE {where_sql} ORDER BY task_id"
            ))
            .unwrap();
        s.query_map([], |row| {
            let values = (0..15)
                .map(|i| row.get::<_, rusqlite::types::Value>(i))
                .collect::<Result<Vec<_>, _>>()?;
            Ok(format!("{values:?}"))
        })
        .unwrap()
        .map(Result::unwrap)
        .collect()
    }
    fn count(&self, predicate: &str) -> usize {
        self.conn()
            .query_row(
                &format!("SELECT count(*) FROM semantic_outbox WHERE {predicate}"),
                [],
                |r| r.get(0),
            )
            .unwrap()
    }
    fn audit(&self, expected: usize) {
        let c = self.conn();
        let (count, distinct, invalid): (usize, usize, usize) = c
            .query_row(
                "SELECT count(*),count(DISTINCT task_id),
             coalesce(sum(space_id!='target' OR attempts_before!=3 OR attempts_after!=3 OR
               token_after IS NOT NULL OR deadline_after IS NOT NULL OR owner_after IS NOT NULL),0)
             FROM reclaim_audit",
                [],
                |r| Ok((r.get(0)?, r.get(1)?, r.get(2)?)),
            )
            .unwrap();
        assert_eq!((count, distinct, invalid), (expected, expected, 0));
    }
}
impl Drop for Fixture {
    fn drop(&mut self) {
        let _ = std::fs::remove_dir_all(&self.root);
    }
}

const PROTECTED: &str = "space_id!='target' OR doc_key NOT LIKE 'expired-%'";

fn worker_round(
    f: &Fixture,
    parallel: bool,
    batch: usize,
    fence: Option<&LifecycleFence>,
) -> usize {
    let limits = WorkerLimits::validated(batch, 60.0, 3600.0, 16).unwrap();
    let handler = |_: &cc_semantic::queue::LeaseGuard<'_>| {
        Ok(TaskExit::NeedsRetry {
            reason: "fixture deliberate durable backoff".into(),
        })
    };
    let report = if parallel {
        drain_pending_parallel_with_lifecycle(&f.db, "new-worker", &limits, fence, 4, &handler)
    } else {
        drain_pending_with_lifecycle(&f.db, "new-worker", &limits, fence, &mut { handler })
    }
    .unwrap();
    assert_eq!(report.claimed, report.retried);
    assert_eq!((report.completed, report.lease_lost), (0, 0));
    report.claimed
}
fn assert_worker_convergence(parallel: bool) {
    let f = Fixture::new("embed");
    let protected = f.rows(PROTECTED);
    let generation = f.db.reads().read_generation().unwrap();
    let mut total = 0;
    for expected in [64, 64, 9, 0] {
        assert_eq!(worker_round(&f, parallel, 128, None), expected);
        total += expected;
        f.audit(total);
        assert_eq!(f.rows(PROTECTED), protected);
        assert_eq!(f.db.reads().read_generation().unwrap(), generation);
        assert_eq!(
            f.count("space_id='target' AND doc_key LIKE 'expired-%' AND state='claimed' AND attempt_count=3"),
            EXPIRED-total
        );
        assert_eq!(
            f.count("space_id='target' AND doc_key LIKE 'expired-%' AND state='pending' AND attempt_count=4 AND lease_token IS NULL AND claim_owner IS NULL"),
            total
        );
        println!("opportunistic parallel={parallel} claimed={expected} total={total} foreign_unchanged=193 attempts_reclaim_delta=0 epoch_unchanged=true");
    }
}
#[test]
fn serial_drain_one_target_page_converges_past_older_foreign_backlog() {
    assert_worker_convergence(false);
}
#[test]
fn four_workers_share_one_target_page_and_converge_without_foreign_starvation() {
    assert_worker_convergence(true);
}
#[test]
fn revocation_drain_one_target_page_converges_and_preserves_other_leases() {
    let f = Fixture::new("revoke");
    let protected = f.rows(PROTECTED);
    let generation = f.db.reads().read_generation().unwrap();
    let mut total = 0;
    for expected in [64, 64, 9, 0] {
        let report =
            drain_space_revocations(&f.db, TARGET, "new-revoker", 60.0, 3600.0, 16, 128).unwrap();
        assert_eq!((report.claimed, report.revoked), (expected, expected));
        assert_eq!(
            (report.retried, report.lease_lost, report.visible_changes),
            (0, 0, 0)
        );
        total += expected;
        f.audit(total);
        assert_eq!(f.rows(PROTECTED), protected);
        assert_eq!(f.db.reads().read_generation().unwrap(), generation);
        assert_eq!(
            f.count("space_id='target' AND doc_key LIKE 'expired-%' AND state='claimed' AND attempt_count=3"),
            EXPIRED-total
        );
        assert_eq!(
            f.count("space_id='target' AND doc_key LIKE 'expired-%' AND state='done' AND attempt_count=4"),
            total
        );
        println!("opportunistic revocation claimed={expected} total={total} foreign_unchanged=193 epoch_unchanged=true");
    }
}
#[test]
fn small_claim_budget_caps_all_three_reclaim_paths() {
    for parallel in [false, true] {
        let f = Fixture::new("embed");
        assert_eq!(worker_round(&f, parallel, 3, None), 3);
        f.audit(3);
        assert_eq!(
            f.count("space_id='target' AND doc_key LIKE 'expired-%' AND state='claimed'"),
            134
        );
    }
    let f = Fixture::new("revoke");
    let report =
        drain_space_revocations(&f.db, TARGET, "new-revoker", 60.0, 3600.0, 16, 3).unwrap();
    assert_eq!(report.revoked, 3);
    f.audit(3);
}
#[test]
fn closed_lifecycle_or_missing_active_space_does_no_reclaim() {
    for parallel in [false, true] {
        let f = Fixture::new("embed");
        let before = f.rows("1=1");
        let generation = f.db.reads().read_generation().unwrap();
        let fence = LifecycleFence::default();
        fence.close();
        assert_eq!(worker_round(&f, parallel, 128, Some(&fence)), 0);
        f.audit(0);
        assert_eq!(f.rows("1=1"), before);
        f.conn()
            .execute(
                "UPDATE semantic_spaces SET state='revoked' WHERE space_id='target'",
                [],
            )
            .unwrap();
        assert_eq!(worker_round(&f, parallel, 128, None), 0);
        f.audit(0);
        assert_eq!(f.rows("1=1"), before);
        assert_eq!(f.db.reads().read_generation().unwrap(), generation);
    }
}
#[test]
fn scoped_db_page_preserves_attempts_and_reports_only_target_exhaustion() {
    let f = Fixture::new("embed");
    let protected = f.rows(PROTECTED);
    let generation = f.db.reads().read_generation().unwrap();
    for (expected, exhausted, total) in [
        (64, false, 64),
        (64, false, 128),
        (9, true, 137),
        (0, true, 137),
    ] {
        let report =
            f.db.reclaim_expired_semantic_space_bounded(TARGET, 64)
                .unwrap();
        assert_eq!((report.reclaimed, report.exhausted), (expected, exhausted));
        f.audit(total);
        assert_eq!(f.rows(PROTECTED), protected);
        assert_eq!(f.db.reads().read_generation().unwrap(), generation);
        assert_eq!(f.count("space_id='target' AND attempt_count!=3"), 0);
    }
}
#[test]
fn scoped_db_reclaim_error_rolls_back_whole_page_and_leaves_connection_usable() {
    let f = Fixture::new("embed");
    let before = f.rows("1=1");
    let generation = f.db.reads().read_generation().unwrap();
    f.conn()
        .execute_batch(
            "CREATE TRIGGER fail_second_reclaim BEFORE UPDATE ON semantic_outbox
         WHEN OLD.doc_key='expired-1' AND NEW.state='pending'
         BEGIN SELECT RAISE(ABORT,'synthetic reclaim failure'); END;",
        )
        .unwrap();
    let error =
        f.db.reclaim_expired_semantic_space_bounded(TARGET, 64)
            .unwrap_err();
    assert!(error.to_string().contains("synthetic reclaim failure"));
    f.audit(0);
    assert_eq!(f.rows("1=1"), before);
    assert_eq!(f.db.reads().read_generation().unwrap(), generation);
    f.conn()
        .execute_batch("DROP TRIGGER fail_second_reclaim;")
        .unwrap();
    assert_eq!(
        f.db.reclaim_expired_semantic_space_bounded(TARGET, 3)
            .unwrap()
            .reclaimed,
        3
    );
    f.audit(3);
}
#[test]
fn scoped_db_invalid_limits_cannot_become_an_unbounded_sqlite_limit() {
    let f = Fixture::new("embed");
    let before = f.rows("1=1");
    assert!(f
        .db
        .reclaim_expired_semantic_space_bounded(TARGET, 0)
        .is_err());
    if usize::BITS > 63 {
        assert!(f
            .db
            .reclaim_expired_semantic_space_bounded(TARGET, usize::MAX)
            .is_err());
    }
    assert_eq!(f.rows("1=1"), before);
    f.audit(0);
}
#[test]
fn real_clock_expiry_preserves_fresh_lease_and_reclaims_only_after_deadline() {
    let f = Fixture::new("embed");
    f.conn().execute("DELETE FROM semantic_outbox", []).unwrap();
    f.insert(TARGET, "real-clock", "embed", "pending", 0.0);
    let c = f.conn();
    let task = claim_next_on(&c, TARGET, "real-clock-owner", now_unix(), 0.15)
        .unwrap()
        .unwrap();
    // A long second lease is a positive control throughout the real wait.
    f.insert(
        TARGET,
        "long-lease",
        "embed",
        "claimed",
        now_unix() + 3600.0,
    );
    let long = f.rows("doc_key='long-lease'");
    let before = f.db.reads().read_generation().unwrap();
    std::thread::sleep(Duration::from_millis(200));
    assert!(now_unix() > task.lease_expires_at);
    let report =
        f.db.reclaim_expired_semantic_space_bounded(TARGET, 1)
            .unwrap();
    assert_eq!((report.reclaimed, report.exhausted), (1, true));
    assert_eq!(f.rows("doc_key='long-lease'"), long);
    assert_eq!(
        f.count(
            "doc_key='real-clock' AND state='pending' AND attempt_count=4 AND lease_token IS NULL"
        ),
        1
    );
    assert_eq!(f.db.reads().read_generation().unwrap(), before);
}
