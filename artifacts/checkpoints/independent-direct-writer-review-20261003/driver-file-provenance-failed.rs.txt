const SCHEMA: &str = include_str!("../../local/source/crates/cc-db/src/sql/index_v1.sql");
type Catalog = Vec<(String, String, String, Option<String>)>;
fn q(name: &str) -> String { format!("\"{}\"", name.replace('"', "\"\"")) }
fn catalog(c: &Connection) -> Catalog {
    c.prepare("SELECT type,name,tbl_name,sql FROM sqlite_schema ORDER BY type,name").unwrap()
        .query_map([], |r| Ok((r.get(0)?,r.get(1)?,r.get(2)?,r.get(3)?))).unwrap()
        .map(Result::unwrap).collect()
}
fn without_explicit(cat: &Catalog) -> Catalog {
    cat.iter().filter(|(t,_,_,s)| t != "index" || s.is_none()).cloned().collect()
}
fn rows(c: &Connection, sql: &str) -> Vec<Vec<Value>> {
    let mut s = c.prepare(sql).unwrap();
    let count = s.column_count();
    s.query_map([], |r| (0..count).map(|i| r.get::<_,Value>(i)).collect()).unwrap()
        .map(Result::unwrap).collect()
}
fn n(c: &Connection, sql: &str) -> i64 { c.query_row(sql, [], |r| r.get(0)).unwrap() }
fn snapshot(c: &Connection) -> Vec<(String, Vec<Vec<Value>>)> {
    let mut result = vec![];
    for (kind,name,_,_) in catalog(c) {
        if kind != "table" {continue;}
        let sql = if name == "metadata" {
            "SELECT key,value FROM metadata WHERE key != 'index_incarnation' ORDER BY key".into()
        } else {format!("SELECT rowid,* FROM {} ORDER BY rowid",q(&name))};
        result.push((name, rows(c,&sql)));
    }
    result
}
fn unit(path: &str, word: &str) -> FileWriteUnit {
    let text = format!("fn {word}() {{ let value = \"{word}\"; }}");
    let source = SourceSnapshot::new(text.as_bytes());
    let chunk = ChunkRecord { chunk_id:format!("chunk:{path}"),file_path:path.into(), language:Language::Rust,
        chunk_index:0,start_line:1,end_line:1,breadcrumb:"independent".into(), text:text.clone(),
        symbol_name:Some(word.into()),symbol_kind:Some(cc_model::symbol::SymbolKind::Function),token_estimate:12,
        parser_tier:ParserTier::TreeSitter,parser_confidence:1.0,
        source:Some(ChunkSource {source:source.identity().clone(),span:source.whole(),
            slice_digest:source.slice_digest(source.whole()).unwrap(),boundary:"whole".into(),owner:None,signature:None})};
    let doc = DocumentRecord::new(&chunk,&cc_model::identity::bytes_hash(b"independent-policy"),0,"review",Err("synthetic no encoder".into())).unwrap();
    let mut outcome = cc_model::parse::ParseOutcome::default();
    outcome.summary = word.into();
    outcome.chunks = vec![chunk];
    outcome.documents = Some(DocumentBatch {records:vec![doc], delta:Default::default()});
    outcome.symbols.push(serde_json::from_value(serde_json::json!({
        "symbol_id":format!("symbol:{path}"),"file_path":path,"name":word,"kind":"function",
        "start_line":1,"end_line":1,"start_col":0,"end_col":1,"parser_tier":"tree_sitter",
        "parser_confidence":1.0,"is_default_export":false,"symbol_uid":format!("uid:{path}")
    })).unwrap());
    outcome.literal_index.push(cc_model::diagnostic::LiteralRecord {literal_id:format!("literal:{path}"),
        file_path:path.into(), literal:word.into(),literal_kind:"string".into(),line:1,container:None,
        confidence:1.0,enclosing_symbol_uid:None,key_path:None});
    FileWriteUnit {rel_path:path.into(),language:Language::Rust,content_hash:format!("hash:{word}"),
        mtime:17.0,size:text.len() as u64,outcome}
}
fn populate(c: &Connection, units: &[FileWriteUnit]) -> CcResult<()> {
    for u in units { IndexDb::insert_file_data(c,u)?; }
    c.execute("UPDATE files SET indexed_at='review-fixed-clock'",[]).unwrap();
    c.execute_batch("INSERT INTO semantic_manifest SELECT doc_key,doc_version,file_path,'encoding','digest','space','artifact','review-clock','review-incarnation' FROM document_manifest;
        INSERT INTO semantic_spaces VALUES('space','{}','active','review-clock');
        INSERT INTO semantic_outbox(doc_key,doc_version,input_digest,space_id,op,state,available_at,created_at,updated_at)
            SELECT doc_key,doc_version,'digest','space','embed','done',0,'review-clock','review-clock' FROM document_manifest;").unwrap();
    Ok(())
}
fn fts(c: &Connection) {
    for (base,projection,fts_table,cols) in [
        ("symbols","rowid,name,symbol_id,file_path","symbols_fts","rowid,name,symbol_id,file_path"),
        ("files","rowid,file_path","file_paths_fts","rowid,file_path"),
        ("files","rowid,file_path,summary,content_excerpt","files_fts","rowid,file_path,summary,content_excerpt"),
        ("literal_index","rowid,literal_id,file_path,literal,literal_kind","literal_fts","rowid,literal_id,file_path,literal,literal_kind"),
        ("chunks","rowid,chunk_id,file_path,breadcrumb,symbol_name,text","chunks_fts","rowid,chunk_id,file_path,breadcrumb,symbol_name,text")
    ] {
        assert_eq!(rows(c,&format!("SELECT {projection} FROM {base} ORDER BY rowid")),
            rows(c,&format!("SELECT {cols} FROM {fts_table} ORDER BY rowid")),"FTS drift {fts_table}");
    }
}
fn matches(c: &Connection, word: &str, want: i64) {
    for table in ["chunks_fts","files_fts","literal_fts","symbols_fts"] {
        let count:i64 = c.query_row(&format!("SELECT count(*) FROM {table} WHERE {table} MATCH ?"),[word],|r|r.get(0)).unwrap();
        assert_eq!(count,want,"{table} term {word}");
    }
}
fn oracle() -> (Connection,Catalog) {
    let c = Connection::open_in_memory().unwrap(); c.execute_batch(SCHEMA).unwrap();
    let cat = catalog(&c); (c,cat)
}
fn canonical(dir: &Path) {
    let (_, expected) = oracle();
    let units = [unit("src/alpha.rs","amber"),unit("src/beta.rs","violet")];
    let mut staged = vec![];
    let mut final_rows = vec![];
    for direct in [true,false] {
        let path = dir.join(if direct {"direct.sqlite3"} else {"normal.sqlite3"});
        let db = IndexDb::open(&path).unwrap().0;
        db.writes().set_metadata("index_epoch","41").unwrap(); db.writes().set_metadata("evidence_epoch","73").unwrap();
        let old = db.reads().read_generation().unwrap();
        let mut pre = vec![];
        let build = |c: &Connection| -> CcResult<()> {
            assert_eq!(catalog(c),without_explicit(&expected),"callback schema BEFORE any reopen");
            populate(c,&units)?; fts(c); matches(c,"amber",1);
            assert_eq!(n(c,"SELECT count(*) FROM document_manifest"),2);
            assert_eq!(n(c,"SELECT count(*) FROM semantic_manifest"),2);
            // Auto constraint must still enforce uniqueness while explicit indexes are absent.
            let e=c.execute("INSERT INTO symbols(symbol_id,file_path,name,kind,start_line,end_line,symbol_uid) VALUES('dup','src/alpha.rs','dup','function',1,1,'uid:src/alpha.rs')",[]).unwrap_err();
            assert!(e.to_string().contains("UNIQUE constraint failed: symbols.symbol_uid"));
            pre=snapshot(c); Ok(())
        };
        let result = if direct {db.admin().rebuild_with_direct_writer(build)} else {db.admin().rebuild_with_temp_db(build)};
        result.unwrap(); staged.push(pre);
        let c=db.read_conn().unwrap();
        assert_eq!(catalog(&c),expected);
        assert_eq!(db.reads().schema_version().unwrap(),24);
        let g=db.reads().read_generation().unwrap();
        assert_eq!(g.index_epoch,42); assert_eq!(g.evidence_epoch,74);
        assert_ne!(old.incarnation,g.incarnation); assert_eq!(g.semantic_epoch,None);
        fts(&c); matches(&c,"amber",1);
        assert_eq!(n(&c,"SELECT count(*) FROM file_paths_fts WHERE file_paths_fts MATCH 'alpha'"),1);
        final_rows.push(snapshot(&c)); drop(c);
        let raw=Connection::open(&path).unwrap();
        raw.execute_batch("INSERT INTO symbols(symbol_id,file_path,name,kind,start_line,end_line) VALUES('extra','src/alpha.rs','cobalt','function',1,1);
            UPDATE symbols SET name='scarlet' WHERE symbol_id='extra';").unwrap();
        assert_eq!(n(&raw,"SELECT count(*) FROM symbols_fts WHERE symbols_fts MATCH 'cobalt'"),0);
        assert_eq!(n(&raw,"SELECT count(*) FROM symbols_fts WHERE symbols_fts MATCH 'scarlet'"),1);
        fts(&raw);
        raw.execute("DELETE FROM symbols WHERE symbol_id='extra'",[]).unwrap(); fts(&raw);
        // Raw file-path trigger update: change and restore before comparing all five projections.
        raw.execute("UPDATE files SET file_path='src/renamed.rs' WHERE file_path='src/alpha.rs'",[]).unwrap();
        assert_eq!(n(&raw,"SELECT count(*) FROM file_paths_fts WHERE file_paths_fts MATCH 'renamed'"),1);
        assert_eq!(n(&raw,"SELECT count(*) FROM file_paths_fts WHERE file_paths_fts MATCH 'alpha'"),0);
        raw.execute("UPDATE files SET file_path='src/alpha.rs' WHERE file_path='src/renamed.rs'",[]).unwrap(); fts(&raw);
        drop(raw);
        let replacement=unit("src/alpha.rs","golden");
        db.writes().replace_files_batch(&[replacement]).unwrap();
        { let c=db.read_conn().unwrap(); fts(&c); matches(&c,"amber",0); matches(&c,"golden",1); }
        db.writes().remove_files_batch(&["src/alpha.rs".into(),"src/beta.rs".into()]).unwrap();
        let c=db.read_conn().unwrap(); fts(&c);
        for table in ["files","chunks","symbols","literal_index","document_manifest","semantic_manifest"] {
            assert_eq!(n(&c,&format!("SELECT count(*) FROM {table}")),0,"dangling {table}");
        }
        assert_eq!(rows(&c,"PRAGMA integrity_check"),vec![vec![Value::Text("ok".into())]]);
    }
    assert_eq!(staged[0],staged[1],"direct vs normal callback data");
    assert_eq!(final_rows[0],final_rows[1],"direct vs normal published rows (only incarnation excluded)");
    // Standalone canonical completed file examined via raw SQLite, with no IndexDb reopen.
    let path=dir.join("standalone-canonical.sqlite3");
    writer::DirectWriter::write_db(&path,SCHEMA,|c| {
        assert_eq!(catalog(c),without_explicit(&expected)); populate(c,&units)?; fts(c); Ok(())
    }).unwrap();
    let c=Connection::open(path).unwrap(); assert_eq!(catalog(&c),expected); fts(&c);
    println!("PASS canonical: base/FTS/documents/semantic; callback and standalone catalog; normal comparison; production replace/delete; objects={}",expected.len());
}
fn helpers(_dir: &Path) {
    let (c,expected)=oracle();
    let drops=writer::drop_index_statements(SCHEMA);
    c.execute_batch(&drops).unwrap(); assert_eq!(catalog(&c),without_explicit(&expected));
    let creates=writer::extract_index_statements(SCHEMA);
    c.execute_batch(&creates).unwrap(); assert_eq!(catalog(&c),expected);
    // A complete trigger has a CREATE INDEX-looking substring after an internal semicolon.
    let sql="CREATE TABLE t(v TEXT); CREATE TABLE a(v TEXT); CREATE TRIGGER tr AFTER INSERT ON t BEGIN INSERT INTO a VALUES('prefix;CREATE INDEX forged ON a(v);suffix'); INSERT INTO a VALUES(CASE WHEN new.v='x' THEN 'a;b' ELSE 'c' END); END; CREATE INDEX real_idx ON t(v)";
    let c=Connection::open_in_memory().unwrap(); c.execute_batch(sql).unwrap();
    let expected=catalog(&c); let drop=writer::drop_index_statements(sql);
    assert!(drop.contains("real_idx")); assert!(!drop.contains("forged"));
    c.execute_batch(&drop).unwrap();
    let extracted=writer::extract_index_statements(sql); assert!(!extracted.contains("forged"),"trigger body leaked into indexes");
    c.execute_batch(&extracted).unwrap(); assert_eq!(catalog(&c),expected);
    println!("PASS helpers: exact canonical drop/create set; automatic constraints retained; complete trigger and no-semicolon final index");
}
const COMPLEX: &str = r#"
/* indépendant ; block */ CREATE TABLE "src;α" (id INTEGER PRIMARY KEY, value TEXT UNIQUE, grp TEXT);
CREATE TABLE [audit;log] (seq INTEGER PRIMARY KEY, msg TEXT);
CREATE UNIQUE INDEX "ON; doubled"" identifier" ON "src;α"(grp) WHERE grp IS NOT NULL;
CREATE INDEX `quote; index` ON [audit;log](msg);
CREATE TRIGGER "tr;α" AFTER INSERT ON "src;α" BEGIN
  INSERT INTO [audit;log](msg) VALUES(CASE WHEN new.id=1 THEN 'seed;it''s' ELSE 'bulk;else' END);
  -- ignored ; CREATE INDEX decoy
  INSERT INTO [audit;log](msg) VALUES('second;body'); /* END; ; */
END;
INSERT INTO "src;α" VALUES(1,'seed','group');
CREATE VIEW "view;α" AS SELECT id,value FROM "src;α" WHERE value != 'ignored;quote';
INSERT INTO [audit;log](msg) VALUES('tail;without delimiter')
"#;
fn synthetic(dir: &Path) {
    let expected=Connection::open_in_memory().unwrap(); expected.execute_batch(COMPLEX).unwrap();
    let schema=catalog(&expected); let seed=snapshot(&expected);
    let p=dir.join("quotes.sqlite3");
    writer::DirectWriter::write_db(&p,COMPLEX,|c| {
        assert_eq!(catalog(c),without_explicit(&schema));
        assert_eq!(snapshot(c),seed,"seed/trigger original execution order");
        // Explicit UNIQUE is absent during bulk; automatic UNIQUE remains.
        c.execute("INSERT INTO \"src;α\" VALUES(2,'bulk',NULL)",[]).unwrap();
        assert!(c.execute("INSERT INTO \"src;α\" VALUES(3,'seed',NULL)",[]).unwrap_err().to_string().contains("UNIQUE"));
        Ok(())
    }).unwrap();
    expected.execute("INSERT INTO \"src;α\" VALUES(2,'bulk',NULL)",[]).unwrap();
    let c=Connection::open(&p).unwrap(); assert_eq!(catalog(&c),schema); assert_eq!(snapshot(&c),snapshot(&expected));
    assert_eq!(n(&c,"SELECT count(*) FROM \"view;α\""),2);
    assert!(c.execute("INSERT INTO \"src;α\" VALUES(4,'new','group')",[]).unwrap_err().to_string().contains("UNIQUE"));
    println!("PASS synthetic: multistatement CASE trigger; Unicode/quotes/comments/semicolon names; seed order; auto constraints; view; quoted explicit indexes restored");
}
fn invalid(dir: &Path) {
    for (i,sql) in [
        "CREATE TABLE t(x); CREATE TRIGGER tr AFTER INSERT ON t BEGIN INSERT INTO t VALUES(1);",
        "CREATE TABLE t(x); INSERT INTO t VALUES('unterminated);",
        "CREATE TABLE t(x); CREATE INDEX bad ON absent(x);",
        "CREATE TABLE t(x); CREATE UNIQUE INDEX u ON t(x); INSERT INTO t VALUES(1),(1);",
        "CREATE TABLE t(x UNIQUE); INSERT INTO t VALUES(1),(1);"
    ].iter().enumerate() {
        let called=Cell::new(false);
        let r=writer::DirectWriter::write_db(&dir.join(format!("invalid-{i}.sqlite3")),sql,|_|{called.set(true);Ok(())});
        assert!(r.is_err(),"invalid accepted {i}"); assert!(!called.get(),"invalid invoked callback {i}");
        println!("EXPECTED invalid-{i}: {}",r.unwrap_err());
    }
    for (i,sql) in ["CREATE TABLE t(x); -- trailing ; comment", "CREATE TABLE t(x); /* comment ;", "CREATE TABLE t(x)"].iter().enumerate() {
        let p=dir.join(format!("tail-{i}.sqlite3")); writer::DirectWriter::write_db(&p,sql,|_|Ok(())).unwrap();
        assert_eq!(n(&Connection::open(p).unwrap(),"SELECT count(*) FROM sqlite_schema WHERE name='t'"),1);
    }
    println!("PASS invalid/tails: expected API errors; callback never called; SQLite-accepted comment and missing delimiter tails retained");
}
fn nul(dir: &Path) {
    for (i,sql) in ["CREATE TABLE prefix(x);\0CREATE TABLE suffix(x);", "-- c\0\nCREATE TABLE suffix(x);", "CREATE TABLE t(x); INSERT INTO t VALUES('a\0b');"].iter().enumerate() {
        let called=Cell::new(false); let p=dir.join(format!("nul-{i}.sqlite3"));
        let e=writer::DirectWriter::write_db(&p,sql,|_|{called.set(true);Ok(())}).expect_err("NUL silently accepted");
        assert!(e.to_string().contains("embedded NUL")); assert!(!called.get()); assert!(!p.exists());
    }
    println!("PASS NUL: suffix/comment/string all rejected before opening path or invoking callback");
}
fn failures(dir: &Path) {
    let p=dir.join("protected-live.sqlite3"); let db=IndexDb::open(&p).unwrap().0;
    db.writes().replace_files_batch(&[unit("src/live.rs","healthy")]).unwrap();
    db.writes().set_metadata("review-marker","preserve-me").unwrap();
    let generation=db.reads().read_generation().unwrap();
    let before=snapshot(&db.read_conn().unwrap());
    let e=db.admin().rebuild_with_direct_writer(|_|Err(CcError::Database("independent callback negative".into()))).unwrap_err();
    assert!(e.to_string().contains("independent callback negative"));
    assert_eq!(snapshot(&db.read_conn().unwrap()),before); assert_eq!(db.reads().read_generation().unwrap(),generation);
    let called=Cell::new(false);
    let e=db.admin().rebuild_with_direct_writer(|c| {
        called.set(true);
        assert_eq!(n(c,"SELECT count(*) FROM sqlite_schema WHERE name='semantic_outbox_live_per_doc'"),0);
        c.execute_batch("INSERT INTO semantic_outbox(doc_key,doc_version,input_digest,space_id,op,state,available_at,created_at,updated_at) VALUES('d','v','i','s','embed','pending',0,'c','u'),('d','v','i','s','embed','claimed',0,'c','u');").unwrap();
        Ok(())
    }).unwrap_err();
    assert!(called.get()); assert!(e.to_string().contains("indexes: UNIQUE constraint failed"),"{e}");
    assert_eq!(snapshot(&db.read_conn().unwrap()),before); assert_eq!(db.reads().read_generation().unwrap(),generation);
    fts(&db.read_conn().unwrap()); matches(&db.read_conn().unwrap(),"healthy",1);
    println!("EXPECTED production delayed unique failure: {e}; live logical rows/catalog/generation unchanged");
    db.admin().rebuild_with_direct_writer(|c|IndexDb::insert_file_data(c,&unit("src/retry.rs","recovered"))).unwrap();
    matches(&db.read_conn().unwrap(),"healthy",0); matches(&db.read_conn().unwrap(),"recovered",1); fts(&db.read_conn().unwrap());
    let p=dir.join("standalone-conflict.sqlite3");
    let e=writer::DirectWriter::write_db(&p,"CREATE TABLE t(x); CREATE UNIQUE INDEX u ON t(x);",|c| {
        c.execute("INSERT INTO t VALUES(1),(1)",[]).unwrap(); Ok(())
    }).unwrap_err();
    assert!(e.to_string().contains("indexes: UNIQUE constraint failed"));
    println!("EXPECTED standalone delayed unique failure: {e}; path_exists={} (staging API error, no live publication claim)",p.exists());
    // Document cleanup trigger must work even with FK checks off, independently of cascades.
    let p=dir.join("document-delete.sqlite3");
    writer::DirectWriter::write_db(&p,SCHEMA,|c|IndexDb::insert_file_data(c,&unit("src/document.rs","document"))).unwrap();
    let c=Connection::open(p).unwrap(); c.execute_batch("PRAGMA foreign_keys=OFF; DELETE FROM chunks;").unwrap();
    assert_eq!(n(&c,"SELECT count(*) FROM document_manifest"),0);
    println!("PASS failures: callback error and explicit partial UNIQUE conflict do not publish; retry succeeds; standalone contract separate; document trigger with FK OFF");
}
fn main() {
    let args:Vec<_>=std::env::args().collect(); let case=&args[1]; let dir=PathBuf::from(&args[2]);
    std::fs::create_dir_all(&dir).unwrap();
    println!("SQLite {} case={case}",rusqlite::version());
    match case.as_str() {"canonical"=>canonical(&dir),"helpers"=>helpers(&dir),"synthetic"=>synthetic(&dir),
        "invalid"=>invalid(&dir),"nul"=>nul(&dir),"failures"=>failures(&dir),_=>panic!("unknown case")}
}
